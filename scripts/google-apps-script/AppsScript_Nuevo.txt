function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Hoja 1") || ss.getSheets()[0];
  var rows = sheet.getDataRange().getValues();
  var partyId = e.parameter.id;
  
  if (!partyId) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "No ID provided" })).setMimeType(ContentService.MimeType.JSON);
  }
  
  var matchingRows = [];
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] && rows[i][0].toString().trim().toLowerCase() === partyId.trim().toLowerCase()) {
      matchingRows.push({ rowIndex: i + 1, rowData: rows[i] });
    }
  }
  
  if (matchingRows.length === 0) {
    return ContentService.createTextOutput(JSON.stringify({ status: "not_found" })).setMimeType(ContentService.MimeType.JSON);
  }
  
  var firstRow = matchingRows[0].rowData;
  var allIntegrantes = [];
  var integrantesData = [];
  var estadoSabadoList = [];
  var estadoCivilList = [];
  var alergiasList = [];
  
  matchingRows.forEach(function(item) {
    var memberName = item.rowData[2] ? item.rowData[2].toString().trim() : "";
    var customGuestId = item.rowData[14] ? item.rowData[14].toString().trim() : "";
    
    // Si la celda O (ID Arcade) está vacía, autogeneramos un ID indestructible basado en la fila
    var guestId = customGuestId !== "" ? customGuestId : ("GID_" + item.rowData[0].toString().trim() + "_" + item.rowIndex);

    if (memberName) {
      allIntegrantes.push(memberName);
      integrantesData.push({ id: guestId, name: memberName });

      if (item.rowData[5]) estadoSabadoList.push(memberName + ": " + item.rowData[5]);
      if (item.rowData[6]) estadoCivilList.push(memberName + ": " + item.rowData[6]);
      if (item.rowData[7] && item.rowData[7].toString().trim() !== "" && item.rowData[7].toString().trim() !== "Ninguna" && item.rowData[7].toString().trim() !== "Nones") {
        alergiasList.push(memberName + ": " + item.rowData[7]);
      }
    }
  });
  
  var headers = rows[0] || [];
  var colPerroNombres = -1;
  var colPerroFotos = -1;
  for (var h = 0; h < headers.length; h++) {
    var headerText = (headers[h] || "").toString().trim().toLowerCase();
    var hasPerr = headerText.indexOf("perr") !== -1 || headerText.indexOf("mascot") !== -1 || headerText.indexOf("dog") !== -1;
    var hasFoto = headerText.indexOf("foto") !== -1 || headerText.indexOf("drive") !== -1 || headerText.indexOf("link") !== -1 || headerText.indexOf("url") !== -1 || headerText.indexOf("imagen") !== -1;
    
    if (hasFoto && hasPerr) {
      colPerroFotos = h;
    } else if (hasPerr) {
      colPerroNombres = h;
    }
  }

  var savedPerros = (colPerroNombres !== -1 && firstRow[colPerroNombres]) ? firstRow[colPerroNombres].toString().trim() : "";
  var savedFotos = (colPerroFotos !== -1 && firstRow[colPerroFotos]) ? firstRow[colPerroFotos].toString().trim() : "";

  var data = {
    status: "success",
    id: firstRow[0],
    nombre: firstRow[1],
    integrantes: allIntegrantes.join(", "),
    integrantesData: integrantesData,
    invitaCivil: firstRow[3],
    pasesSabado: firstRow[4],
    estadoSabado: estadoSabadoList.length > 0 ? estadoSabadoList.join(" | ") : firstRow[5],
    estadoCivil: estadoCivilList.length > 0 ? estadoCivilList.join(" | ") : firstRow[6],
    alergias: alergiasList.length > 0 ? alergiasList.join(" | ") : "",
    celebracion: firstRow[8],
    perros: savedPerros,
    fotosPerros: savedFotos
  };
  
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Hoja 1") || ss.getSheets()[0];
    
    // --- 🎮 ESTADÍSTICAS DEL ARCADE EN LA PESTAÑA PRINCIPAL ---
    if (contents.action === "updateGameStats") {
      var rows = sheet.getDataRange().getValues();
      var rowIndex = -1;
      
      // Buscar la fila exacta del invitado por su Guest ID
      if (contents.guestId) {
        for (var i = 1; i < rows.length; i++) {
          var rowCustomId = rows[i][14] ? rows[i][14].toString().trim() : "";
          var rowAutoId = "GID_" + rows[i][0].toString().trim() + "_" + (i + 1);
          
          if (rowCustomId === contents.guestId || rowAutoId === contents.guestId) {
            rowIndex = i + 1;
            break;
          }
        }
      }
      
      // Fallback de seguridad: buscar por nombre si no se mandó el ID
      if (rowIndex === -1 && contents.realName) {
        for (var i = 1; i < rows.length; i++) {
          if (rows[i][2] && rows[i][2].toString().trim().toLowerCase() === contents.realName.trim().toLowerCase()) {
            rowIndex = i + 1;
            break;
          }
        }
      }
      
      if (rowIndex > -1) {
        // Actualizar Columnas: L (12) GamerTag, M (13) ChiCoins, N (14) Tiempo Jugado
        sheet.getRange(rowIndex, 12).setValue(contents.gamerTag);
        sheet.getRange(rowIndex, 13).setValue(contents.totalChiCoins);
        sheet.getRange(rowIndex, 14).setValue(contents.timePlayedSeconds);
      }
      
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }
    // --- FIN DE LÓGICA ARCADE ---


    // --- LÓGICA DE RSVP ---
    var partyId = contents.id;
    var rowsData = sheet.getDataRange().getValues();
    
    var matchingRowsInfo = [];
    var familyName = "";
    
    for (var i = 1; i < rowsData.length; i++) {
      if (rowsData[i][0] && rowsData[i][0].toString().trim().toLowerCase() === partyId.trim().toLowerCase()) {
        matchingRowsInfo.push({
          rowIndex: i + 1,
          memberName: rowsData[i][2] ? rowsData[i][2].toString().trim() : ""
        });
        if (!familyName) familyName = rowsData[i][1];
      }
    }
    
    if (matchingRowsInfo.length === 0) {
      return ContentService.createTextOutput(JSON.stringify({ status: "not_found" })).setMimeType(ContentService.MimeType.JSON);
    }

    function parseMemberValue(fullStr, memberName) {
      if (!fullStr) return "";
      var parts = fullStr.split("|");
      for (var p = 0; p < parts.length; p++) {
        var pair = parts[p].split(":");
        if (pair.length >= 2) {
          var nameKey = pair[0].trim();
          var val = pair.slice(1).join(":").trim();
          if (nameKey.toLowerCase() === memberName.toLowerCase() || memberName.toLowerCase().indexOf(nameKey.toLowerCase()) !== -1 || nameKey.toLowerCase().indexOf(memberName.toLowerCase()) !== -1) {
            return val;
          }
        }
      }
      return "";
    }

    // 1. Asignar SI / NO y notas limpias por renglón
    if (matchingRowsInfo.length > 1) {
      matchingRowsInfo.forEach(function(info) {
        var sabadoVal = parseMemberValue(contents.estadoSabado, info.memberName) || "SI";
        var civilVal = parseMemberValue(contents.estadoCivil, info.memberName) || "N/A";
        var rawAlergia = parseMemberValue(contents.alergias, info.memberName);
        var alergiaVal = (rawAlergia && rawAlergia.toString().trim().toLowerCase() !== "ninguna" && rawAlergia.toString().trim().toLowerCase() !== "n/a") ? rawAlergia.toString().trim() : "";

        sheet.getRange(info.rowIndex, 6).setValue(sabadoVal);
        sheet.getRange(info.rowIndex, 7).setValue(civilVal);
        sheet.getRange(info.rowIndex, 8).setValue(alergiaVal);
        sheet.getRange(info.rowIndex, 9).setValue(contents.celebracion || "");
      });
    } else {
      var singleSabado = parseMemberValue(contents.estadoSabado, matchingRowsInfo[0].memberName) || contents.estadoSabado || "SI";
      var singleCivil = parseMemberValue(contents.estadoCivil, matchingRowsInfo[0].memberName) || contents.estadoCivil || "N/A";
      var rawSingleAlergia = parseMemberValue(contents.alergias, matchingRowsInfo[0].memberName) || contents.alergias || "";
      var singleAlergia = (rawSingleAlergia && rawSingleAlergia.toString().trim().toLowerCase() !== "ninguna" && rawSingleAlergia.toString().trim().toLowerCase() !== "n/a") ? rawSingleAlergia.toString().trim() : "";

      sheet.getRange(matchingRowsInfo[0].rowIndex, 6).setValue(singleSabado);
      sheet.getRange(matchingRowsInfo[0].rowIndex, 7).setValue(singleCivil);
      sheet.getRange(matchingRowsInfo[0].rowIndex, 8).setValue(singleAlergia);
      sheet.getRange(matchingRowsInfo[0].rowIndex, 9).setValue(contents.celebracion || "");
    }

    // 2. Procesar perrihijos (Club Secreto de Maddie: Fotos a Drive y Nombres al Sheet)
    var dogNamesList = [];
    var dogLinksList = [];
    if (contents.dogs && contents.dogs.length > 0) {
      try {
        var folderName = "🐾 Perrihijos Boda Chicos";
        var folders = DriveApp.getFoldersByName(folderName);
        var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

        contents.dogs.forEach(function(dog, idx) {
          var dogName = (dog.name || "").toString().trim() || ("Perrito " + (idx + 1));
          dogNamesList.push(dogName);

          if (dog.photoBase64) {
            try {
              var splitBase = dog.photoBase64.split(",");
              var rawData = splitBase.length > 1 ? splitBase[1] : splitBase[0];
              var contentType = "image/jpeg";
              if (dog.photoBase64.indexOf("data:") === 0) {
                contentType = dog.photoBase64.substring(5, dog.photoBase64.indexOf(";"));
              }
              var decoded = Utilities.base64Decode(rawData);
              var safeGuest = (familyName || partyId).replace(/[^a-zA-Z0-9áéíóúÁÉÍÓÚñÑ\s-_]/g, "").trim();
              var safeDog = dogName.replace(/[^a-zA-Z0-9áéíóúÁÉÍÓÚñÑ\s-_]/g, "").trim();
              var fileName = partyId + " - " + safeGuest + " - Perro " + safeDog + ".jpg";

              var blob = Utilities.newBlob(decoded, contentType, fileName);
              var file = folder.createFile(blob);
              file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
              dogLinksList.push(file.getUrl());
            } catch (fileErr) {
              Logger.log("Error al guardar archivo en Drive: " + fileErr.toString());
            }
          }
        });
      } catch (driveErr) {
        Logger.log("Error al acceder a Google Drive: " + driveErr.toString());
      }
    }

    var dogNamesStr = dogNamesList.join(", ");
    var dogLinksStr = dogLinksList.join(", ");

    if (dogNamesStr !== "" || dogLinksStr !== "") {
      // Buscar columnas dinámicas para Perros / Fotos de Perros en el encabezado (Fila 1)
      var headers = rowsData[0] || [];
      var colPerroNombres = -1;
      var colPerroFotos = -1;

      for (var h = 0; h < headers.length; h++) {
        var headerText = (headers[h] || "").toString().trim().toLowerCase();
        var hasPerr = headerText.indexOf("perr") !== -1 || headerText.indexOf("mascot") !== -1 || headerText.indexOf("dog") !== -1;
        var hasFoto = headerText.indexOf("foto") !== -1 || headerText.indexOf("drive") !== -1 || headerText.indexOf("link") !== -1 || headerText.indexOf("url") !== -1 || headerText.indexOf("imagen") !== -1;
        
        if (hasFoto && hasPerr) {
          colPerroFotos = h + 1;
        } else if (hasPerr) {
          colPerroNombres = h + 1;
        }
      }

      // Si no existen en los encabezados, crearlos al final para no pisar ninguna columna existente
      if (colPerroNombres === -1) {
        colPerroNombres = sheet.getLastColumn() + 1;
        sheet.getRange(1, colPerroNombres).setValue("🐾 Perrihijos");
      }
      if (colPerroFotos === -1) {
        colPerroFotos = sheet.getLastColumn() + 1;
        sheet.getRange(1, colPerroFotos).setValue("📷 Fotos Perros (Drive)");
      }

      matchingRowsInfo.forEach(function(info) {
        if (dogNamesStr !== "") sheet.getRange(info.rowIndex, colPerroNombres).setValue(dogNamesStr);
        if (dogLinksStr !== "") sheet.getRange(info.rowIndex, colPerroFotos).setValue(dogLinksStr);
      });
    }

    // 3. Calcular analíticos globales precisos
    SpreadsheetApp.flush();
    var updatedRows = sheet.getDataRange().getValues();
    var confirmadosSabado = 0;
    var declinadosSabado = 0;
    var pendientes = 0;

    for (var j = 1; j < updatedRows.length; j++) {
      var memberName = updatedRows[j][2] ? updatedRows[j][2].toString().trim() : "";
      if (memberName !== "") { 
        var statusSabado = updatedRows[j][5] ? updatedRows[j][5].toString().trim().toUpperCase() : "";
        if (statusSabado === "") {
          pendientes += 1;
        } else if (statusSabado === "SI" || statusSabado.indexOf(": SI") !== -1) {
          confirmadosSabado += 1;
        } else if (statusSabado === "NO" || statusSabado.indexOf(": NO") !== -1) {
          declinadosSabado += 1;
        }
      }
    }

    // 4. Notificación por correo instantánea con emoji 🍽️
    var recipient = "memoge55@gmail.com, mariana.cordero.p@gmail.com";
    var subject = "💌 Nueva Confirmación RSVP: " + (familyName || partyId);
    
    var body = "¡Hola Mariana y Memo!\n\n" +
               "Se ha recibido una respuesta en la invitación web:\n\n" +
               "📌 Invitación: " + familyName + "\n" +
               "💒 Boda & Fiesta (Sábado): " + contents.estadoSabado + "\n" +
               "🥂 Boda Civil (Viernes): " + (contents.estadoCivil || "N/A") + "\n" +
               "🍽️ Restricciones alimenticias: " + (contents.alergias || "Ninguna") + "\n" +
               "🎉 Celebración especial: " + (contents.celebracion || "Ninguna") + "\n" +
               (dogNamesStr !== "" ? ("🐾 Perrihijo(s): " + dogNamesStr + "\n") : "") +
               (dogLinksStr !== "" ? ("📷 Fotos en Drive: " + dogLinksStr + "\n") : "") + "\n" +
               "─────────────────────────────\n" +
               "📊 RESUMEN GENERAL ACUMULADO:\n" +
               "• Confirmados (Sábado): " + confirmadosSabado + " personas\n" +
               "• Declinados (Sábado): " + declinadosSabado + " personas\n" +
               "• Pendientes por responder: " + pendientes + " personas\n\n" +
               "Ver lista completa en tu Google Sheet:\n" + ss.getUrl();

    MailApp.sendEmail(recipient, subject, body);

    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    Logger.log("Error en doPost: " + err.toString());
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * 🛠️ FUNCIÓN SEGURA: Restaura los datos del gráfico y corrige la tabla de restricciones
 * Apuntando a las pestañas exactas: "Hoja 1" y "Analíticos"
 */
function restaurarYCorregirTablero() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var tableroSheet = ss.getSheetByName("Analíticos");
  
  if (!tableroSheet) {
    var sheets = ss.getSheets();
    for (var s = 0; s < sheets.length; s++) {
      var name = sheets[s].getName().toLowerCase();
      if (sheets[s].getSheetId() === 105527289 || name.indexOf("analit") !== -1) {
        tableroSheet = sheets[s];
        break;
      }
    }
  }
  
  if (!tableroSheet && ss.getSheets().length > 1) {
    tableroSheet = ss.getSheets()[1];
  }
  
  if (!tableroSheet) {
    Logger.log("No se encontró la pestaña Analíticos.");
    return;
  }
  
  // 1. Restaurar las filas 13 y 14 que alimentan el gráfico "Confirmados por Lado (Novio vs Novia)"
  tableroSheet.getRange("A13").setValue("🤵🏻‍♂️ Confirmados Novio (Sí)");
  tableroSheet.getRange("B13").setFormula('=COUNTIFS(\'Hoja 1\'!K:K, "Novio", \'Hoja 1\'!F:F, "SI")');
  
  tableroSheet.getRange("A14").setValue("👰🏻‍♀️ Confirmadas Novia (Sí)");
  tableroSheet.getRange("B14").setFormula('=COUNTIFS(\'Hoja 1\'!K:K, "Novia", \'Hoja 1\'!F:F, "SI")');
  
  // 2. Limpiar filas 15, 16, 17 para eliminar el #REF! y los duplicados
  tableroSheet.getRange("A15:D17").clearContent();
  
  // 3. Fila 18: Banner con el nuevo emoji 🍽️
  tableroSheet.getRange("A18").setValue("🍽️ LISTA DE RESTRICCIONES ALIMENTICIAS & CELEBRACIONES");
  
  // 4. Fila 19: Encabezados limpios
  tableroSheet.getRange("A19").setValue("Invitación");
  tableroSheet.getRange("B19").setValue("Integrante");
  tableroSheet.getRange("C19").setValue("Restricciones Alimenticias");
  tableroSheet.getRange("D19").setValue("Celebración Especial");
  
  // 5. Limpiar datos viejos desplazados a la derecha de la fila 20 en adelante
  tableroSheet.getRange("E20:I100").clearContent();
  
  // 6. Fila 20 (A20): Fórmula QUERY exacta apuntando a 'Hoja 1' para traer B, C, H, I
  var formula = '=QUERY(\'Hoja 1\'!A2:I, "SELECT B, C, H, I WHERE (H IS NOT NULL AND H != \'\' AND H != \'Ninguna\') OR (I IS NOT NULL AND I != \'\')", 0)';
  tableroSheet.getRange("A20").setFormula(formula);
  
  try {
    SpreadsheetApp.getUi().alert("¡Listo! 🍽️ Las referencias a 'Hoja 1' y 'Analíticos' se corrigieron exitosamente.");
  } catch(e) {
    Logger.log("Tablero restaurado con éxito.");
  }
}

function onOpen() {
  try {
    var ui = SpreadsheetApp.getUi();
    ui.createMenu("💍 Boda Mariana & Memo")
      .addItem("🍽️ Restaurar Gráfico y Corregir Tablero", "restaurarYCorregirTablero")
      .addToUi();
  } catch(e) {
    Logger.log("onOpen");
  }
}
